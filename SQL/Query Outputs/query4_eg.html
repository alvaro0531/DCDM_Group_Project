WITH target_genes AS (
  SELECT DISTINCT
    c.mouse_gene_symbol,
    c.mgi_gene_id
  FROM data_base_merged.clean_data c
  WHERE c.mouse_gene_symbol IN ('Rhbdl2', 'Scx', 'Muc5ac', 'Parpbp')
    AND c.mgi_gene_id IS NOT NULL
),
disease_summary AS (
  SELECT
    tg.mgi_gene_id,
    COUNT(DISTINCT dc.disease_row_id) AS n_disease_rows,
    COUNT(DISTINCT gb.mouse_genotype_id) AS n_bridged_genotypes,
    GROUP_CONCAT(
      DISTINCT CONCAT(dc.doid, ': ', COALESCE(hd.disease_name, dc.disease_name))
      ORDER BY dc.doid
      SEPARATOR '; '
    ) AS diseases
  FROM target_genes tg
  INNER JOIN data_base_merged.disease_clean dc
    ON tg.mgi_gene_id = dc.mgi_gene_id
  LEFT JOIN data_base_merged.human_diseases hd
    ON dc.doid = hd.doid
  LEFT JOIN data_base_merged.genotype_bridge gb
    ON dc.disease_row_id = gb.disease_row_id
  GROUP BY tg.mgi_gene_id
),
bespoke_summary AS (
  SELECT
    b.mgi_gene_id,
    COUNT(DISTINCT b.bespoke_row_id) AS n_bespoke_rows,
    MAX(b.in_impc_dataset) AS in_impc_dataset,
    GROUP_CONCAT(DISTINCT b.hp_term_name ORDER BY b.hp_term_name SEPARATOR '; ') AS hp_terms,
    GROUP_CONCAT(
      DISTINCT CONCAT(bmp.mp_term_id, ' (', COALESCE(mt.mp_term_name, 'unnamed'), ')')
      ORDER BY bmp.mp_term_id
      SEPARATOR '; '
    ) AS bespoke_mp_terms
  FROM data_base_merged.bespoke_task_table b
  LEFT JOIN data_base_merged.bespoke_task_mp_bridge bmp
    ON b.bespoke_row_id = bmp.bespoke_row_id
  LEFT JOIN data_base_merged.mp_terms mt
    ON bmp.mp_term_id = mt.mp_term_id
  GROUP BY b.mgi_gene_id
),
lacz_summary AS (
  SELECT
    c.mgi_gene_id,
    COUNT(DISTINCT l.result_id) AS lacz_result_rows,
    SUM(CASE WHEN l.lacz_brain IS NOT NULL AND l.lacz_brain != '' THEN 1 ELSE 0 END) AS lacz_brain_n,
    SUM(CASE WHEN l.lacz_heart IS NOT NULL AND l.lacz_heart != '' THEN 1 ELSE 0 END) AS lacz_heart_n,
    SUM(CASE WHEN l.lacz_liver IS NOT NULL AND l.lacz_liver != '' THEN 1 ELSE 0 END) AS lacz_liver_n,
    SUM(CASE WHEN l.lacz_lung IS NOT NULL AND l.lacz_lung != '' THEN 1 ELSE 0 END) AS lacz_lung_n,
    SUM(CASE WHEN l.lacz_kidney IS NOT NULL AND l.lacz_kidney != '' THEN 1 ELSE 0 END) AS lacz_kidney_n,
    SUM(CASE WHEN l.lacz_skin IS NOT NULL AND l.lacz_skin != '' THEN 1 ELSE 0 END) AS lacz_skin_n,
    SUM(CASE WHEN l.lacz_skeletal_muscle IS NOT NULL AND l.lacz_skeletal_muscle != '' THEN 1 ELSE 0 END) AS lacz_muscle_n,
    SUM(CASE WHEN l.lacz_eye IS NOT NULL AND l.lacz_eye != '' THEN 1 ELSE 0 END) AS lacz_eye_n,
    SUM(CASE WHEN l.lacz_colon IS NOT NULL AND l.lacz_colon != '' THEN 1 ELSE 0 END) AS lacz_colon_n,
    SUM(CASE WHEN l.lacz_blood IS NOT NULL AND l.lacz_blood != '' THEN 1 ELSE 0 END) AS lacz_blood_n,
    SUM(CASE WHEN l.lacz_vascular_system IS NOT NULL AND l.lacz_vascular_system != '' THEN 1 ELSE 0 END) AS lacz_vascular_n
  FROM data_base_merged.clean_data c
  INNER JOIN data_base_merged.lacz_expression l
    ON c.result_id = l.result_id
  WHERE c.mouse_gene_symbol IN ('Rhbdl2', 'Scx', 'Muc5ac', 'Parpbp')
  GROUP BY c.mgi_gene_id
),
chrom2_summary AS (
  SELECT
    cg.mgi_gene_id,
    COUNT(DISTINCT ct.ensembl_transcript_id) AS chrom2_transcripts,
    COUNT(DISTINCT c2.ensembl_exon_id) AS chrom2_exons
  FROM data_base_merged.chrom2_gene cg
  LEFT JOIN data_base_merged.chrom2_transcript ct
    ON cg.ensembl_gene_id = ct.ensembl_gene_id
  LEFT JOIN data_base_merged.chrom2 c2
    ON ct.ensembl_transcript_id = c2.ensembl_transcript_id
  GROUP BY cg.mgi_gene_id
),
phenotype_summary AS (
  SELECT
    c.mgi_gene_id,
    COUNT(DISTINCT c.result_id) AS grouped_tests,
    COUNT(DISTINCT pg.group_id) AS groups_tested,
    SUM(CASE WHEN c.p_value < 0.0001 THEN 1 ELSE 0 END) AS sig_hits,
    ROUND(
      SUM(CASE WHEN c.p_value < 0.0001 THEN 1 ELSE 0 END) / NULLIF(COUNT(DISTINCT c.result_id), 0),
      4
    ) AS hit_rate,
    MIN(c.p_value) AS min_p_value,
    ROUND(-LOG10(GREATEST(MIN(c.p_value), 1e-300)), 2) AS neg_log10_min_p
  FROM data_base_merged.clean_data c
  INNER JOIN data_base_merged.parameter_group_member pgm
    ON c.parameter_stable_id = pgm.parameter_stable_id
  INNER JOIN data_base_merged.parameter_group pg
    ON pgm.group_id = pg.group_id
  WHERE c.p_value IS NOT NULL
  GROUP BY c.mgi_gene_id
)
SELECT
  DENSE_RANK() OVER (ORDER BY COALESCE(ps.sig_hits, 0) DESC, ps.min_p_value ASC) AS cohort_rank,
  tg.mouse_gene_symbol,
  tg.mgi_gene_id,
  COALESCE(ps.grouped_tests, 0) AS grouped_tests,
  COALESCE(ps.groups_tested, 0) AS groups_tested,
  COALESCE(ps.sig_hits, 0) AS sig_hits,
  COALESCE(ps.hit_rate, 0) AS hit_rate,
  ps.min_p_value,
  ps.neg_log10_min_p,
  COALESCE(ds.n_disease_rows, 0) AS n_disease_rows,
  COALESCE(ds.n_bridged_genotypes, 0) AS n_bridged_genotypes,
  ds.diseases,
  COALESCE(bs.n_bespoke_rows, 0) AS n_bespoke_rows,
  COALESCE(bs.in_impc_dataset, 0) AS in_impc_dataset,
  bs.hp_terms,
  bs.bespoke_mp_terms,
  COALESCE(ls.lacz_result_rows, 0) AS lacz_result_rows,
  COALESCE(ls.lacz_brain_n, 0) AS lacz_brain_n,
  COALESCE(ls.lacz_heart_n, 0) AS lacz_heart_n,
  COALESCE(ls.lacz_liver_n, 0) AS lacz_liver_n,
  COALESCE(ls.lacz_lung_n, 0) AS lacz_lung_n,
  COALESCE(ls.lacz_kidney_n, 0) AS lacz_kidney_n,
  COALESCE(ls.lacz_skin_n, 0) AS lacz_skin_n,
  COALESCE(ls.lacz_muscle_n, 0) AS lacz_muscle_n,
  COALESCE(ls.lacz_eye_n, 0) AS lacz_eye_n,
  COALESCE(ls.lacz_colon_n, 0) AS lacz_colon_n,
  COALESCE(ls.lacz_blood_n, 0) AS lacz_blood_n,
  COALESCE(ls.lacz_vascular_n, 0) AS lacz_vascular_n,
  COALESCE(c2s.chrom2_transcripts, 0) AS chrom2_transcripts,
  COALESCE(c2s.chrom2_exons, 0) AS chrom2_exons
FROM target_genes tg
LEFT JOIN phenotype_summary ps
  ON tg.mgi_gene_id = ps.mgi_gene_id
LEFT JOIN disease_summary ds
  ON tg.mgi_gene_id = ds.mgi_gene_id
LEFT JOIN bespoke_summary bs
  ON tg.mgi_gene_id = bs.mgi_gene_id
LEFT JOIN lacz_summary ls
  ON tg.mgi_gene_id = ls.mgi_gene_id
LEFT JOIN chrom2_summary c2s
  ON tg.mgi_gene_id = c2s.mgi_gene_id
ORDER BY cohort_rank, tg.mouse_gene_symbol
