WITH gene_group_stats AS (
  SELECT
    pg.group_id,
    pg.group_name AS parameter_group,
    c.mouse_gene_symbol,
    c.mgi_gene_id,
    COUNT(DISTINCT c.result_id) AS total_tests,
    SUM(CASE WHEN c.p_value < 0.0001 THEN 1 ELSE 0 END) AS sig_hits,
    ROUND(
      SUM(CASE WHEN c.p_value < 0.0001 THEN 1 ELSE 0 END) / NULLIF(COUNT(DISTINCT c.result_id), 0),
      4
    ) AS hit_rate,
    MIN(c.p_value) AS min_p_value,
    ROUND(-LOG10(GREATEST(MIN(c.p_value), 1e-300)), 2) AS neg_log10_min_p
  FROM data_base_merged.clean_data c
  INNER JOIN data_base_merged.parameter_group_member pgm
    ON c.parameter_stable_id = pgm.parameter_stable_id
  INNER JOIN data_base_merged.parameter_group pg
    ON pgm.group_id = pg.group_id
  WHERE c.mouse_gene_symbol IS NOT NULL
    AND c.mgi_gene_id IS NOT NULL
    AND c.p_value IS NOT NULL
  GROUP BY
    pg.group_id,
    pg.group_name,
    c.mouse_gene_symbol,
    c.mgi_gene_id
),
top5_by_group AS (
  SELECT
    *,
    ROW_NUMBER() OVER (
      PARTITION BY group_id
      ORDER BY sig_hits DESC, min_p_value ASC, mouse_gene_symbol ASC
    ) AS group_gene_rank
  FROM gene_group_stats
),
top_genes AS (
  SELECT DISTINCT
    mgi_gene_id
  FROM top5_by_group
  WHERE group_gene_rank <= 5
),
disease_agg AS (
  SELECT
    dc.mgi_gene_id,
    COUNT(DISTINCT dc.doid) AS n_diseases,
    COUNT(DISTINCT gb.mouse_genotype_id) AS n_bridged_genotypes,
    GROUP_CONCAT(
      DISTINCT CONCAT(dc.doid, ': ', COALESCE(hd.disease_name, dc.disease_name))
      ORDER BY dc.doid
      SEPARATOR '; '
    ) AS diseases
  FROM top_genes tg
  INNER JOIN data_base_merged.disease_clean dc
    ON tg.mgi_gene_id = dc.mgi_gene_id
  LEFT JOIN data_base_merged.human_diseases hd
    ON dc.doid = hd.doid
  LEFT JOIN data_base_merged.genotype_bridge gb
    ON dc.disease_row_id = gb.disease_row_id
  GROUP BY dc.mgi_gene_id
),
bespoke_agg AS (
  SELECT
    b.mgi_gene_id,
    MAX(COALESCE(b.in_impc_dataset, 0)) AS in_impc_dataset,
    COUNT(DISTINCT bmp.mp_term_id) AS n_bespoke_mp_terms,
    GROUP_CONCAT(
      DISTINCT CONCAT(bmp.mp_term_id, ' (', COALESCE(mt.mp_term_name, 'unmapped'), ')')
      ORDER BY bmp.mp_term_id
      SEPARATOR '; '
    ) AS bespoke_mp_terms
  FROM top_genes tg
  INNER JOIN data_base_merged.bespoke_task_table b
    ON tg.mgi_gene_id = b.mgi_gene_id
  LEFT JOIN data_base_merged.bespoke_task_mp_bridge bmp
    ON b.bespoke_row_id = bmp.bespoke_row_id
  LEFT JOIN data_base_merged.mp_terms mt
    ON bmp.mp_term_id = mt.mp_term_id
  GROUP BY b.mgi_gene_id
)
SELECT
  t.parameter_group,
  t.group_gene_rank,
  t.mouse_gene_symbol,
  t.mgi_gene_id,
  t.total_tests,
  t.sig_hits,
  t.hit_rate,
  t.min_p_value,
  t.neg_log10_min_p,
  COALESCE(d.n_diseases, 0) AS n_diseases,
  COALESCE(d.n_bridged_genotypes, 0) AS n_bridged_genotypes,
  d.diseases,
  COALESCE(b.in_impc_dataset, 0) AS in_impc_dataset,
  COALESCE(b.n_bespoke_mp_terms, 0) AS n_bespoke_mp_terms,
  b.bespoke_mp_terms
FROM top5_by_group t
LEFT JOIN disease_agg d
  ON t.mgi_gene_id = d.mgi_gene_id
LEFT JOIN bespoke_agg b
  ON t.mgi_gene_id = b.mgi_gene_id
WHERE t.group_gene_rank <= 5
ORDER BY t.parameter_group, t.group_gene_rank
