WITH target_chr2_genes AS (
  SELECT DISTINCT
    cg.gene_name,
    cg.ensembl_gene_id
  FROM data_base_merged.chrom2_gene cg
  WHERE cg.gene_name IN ('Gpr107', 'Lhx2', 'Cdh4', 'Ckmt1')
),
chrom2_chain AS (
  SELECT
    tg.gene_name,
    tg.ensembl_gene_id,
    ct.ensembl_transcript_id,
    c2.ensembl_exon_id
  FROM target_chr2_genes tg
  LEFT JOIN data_base_merged.chrom2_transcript ct
    ON tg.ensembl_gene_id = ct.ensembl_gene_id
  LEFT JOIN data_base_merged.chrom2 c2
    ON ct.ensembl_transcript_id = c2.ensembl_transcript_id
)
SELECT
  gene_name,
  ensembl_gene_id,
  COUNT(DISTINCT ensembl_transcript_id) AS chrom2_transcripts,
  COUNT(DISTINCT ensembl_exon_id) AS chrom2_exons,
  GROUP_CONCAT(DISTINCT ensembl_transcript_id ORDER BY ensembl_transcript_id SEPARATOR ', ') AS transcript_ids,
  SUM(COUNT(DISTINCT ensembl_exon_id)) OVER () AS total_selected_chr2_exons
FROM chrom2_chain
GROUP BY
  gene_name,
  ensembl_gene_id
ORDER BY
  chrom2_exons DESC,
  gene_name ASC
