-- Query 2 Gene and Parameter Group Summary
SELECT
  c.mouse_gene_symbol,
  c.mgi_gene_id,
  pg.group_name AS parameter_group,
  COUNT(DISTINCT c.result_id) AS total_tests,
  SUM(CASE WHEN c.p_value < 0.0001 THEN 1 ELSE 0 END) AS significant_hits,
  MIN(c.p_value) AS min_p_value,
  -LOG10(GREATEST(MIN(c.p_value), 1e-300)) AS neg_log10_min_p,
  MAX(b.in_impc_dataset) AS in_bespoke_impc_overlap,
  COUNT(DISTINCT c2.ensembl_exon_id) AS chrom2_exons
FROM data_base_merged.clean_data c
INNER JOIN data_base_merged.parameter_group_member pgm
  ON c.parameter_stable_id = pgm.parameter_stable_id
INNER JOIN data_base_merged.parameter_group pg
  ON pgm.group_id = pg.group_id
LEFT JOIN data_base_merged.bespoke_task_table b
  ON c.mgi_gene_id = b.mgi_gene_id
LEFT JOIN data_base_merged.chrom2_gene cg
  ON c.mgi_gene_id = cg.mgi_gene_id
LEFT JOIN data_base_merged.chrom2_transcript ct
  ON cg.ensembl_gene_id = ct.ensembl_gene_id
LEFT JOIN data_base_merged.chrom2 c2
  ON ct.ensembl_transcript_id = c2.ensembl_transcript_id
WHERE c.mouse_gene_symbol IN ('Rhbdl2', 'Scx', 'Muc5ac', 'Parpbp')
  AND c.p_value IS NOT NULL
GROUP BY
  c.mouse_gene_symbol,
  c.mgi_gene_id,
  pg.group_name
ORDER BY
  c.mouse_gene_symbol,
  min_p_value ASC
